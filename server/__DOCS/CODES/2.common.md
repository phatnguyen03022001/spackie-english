  2: 
  3: import {
  4:   ArgumentsHost,
  5:   Catch,
  6:   ExceptionFilter,
  7:   HttpException,
  8:   HttpStatus,
  9: } from '@nestjs/common';
 10: import { Request, Response } from 'express';
 11: import { LoggerService } from '../logger/logger.service';
 12: import { ErrorResponseDto } from '../dto/error-response.dto';
 13: import { ERROR_CODES } from '@common/constants';
 14: 
 15: @Catch(HttpException, Error)
 16: export class HttpExceptionFilter implements ExceptionFilter {
 17:   constructor(private readonly logger: LoggerService) {
 18:     this.logger.setContext(HttpExceptionFilter.name);
 19:   }
 20: 
 21:   catch(exception: unknown, host: ArgumentsHost): void {
 22:     const { request, response } = this.getContext(host);
 23: 
 24:     const { status, code, message, details, stack } =
 25:       this.normalizeException(exception);
 26: 
 27:     const errorResponse = new ErrorResponseDto(
 28:       status,
 29:       code,
 30:       message,
 31:       request.url,
 32:       details,
 33:     );
 34: 
 35:     this.logException(
 36:       request,
 37:       status,
 38:       code,
 39:       message,
 40:       exception,
 41:       stack,
 42:       details,
 43:     );
 44: 
 45:     response.status(status).json(errorResponse);
 46:   }
 47: 
 48:   // =====================
 49:   // CONTEXT HELPERS
 50:   // =====================
 51:   private getContext(host: ArgumentsHost) {
 52:     const ctx = host.switchToHttp();
 53: 
 54:     return {
 55:       request: ctx.getRequest<Request>(),
 56:       response: ctx.getResponse<Response>(),
 57:     };
 58:   }
 59: 
 60:   // =====================
 61:   // NORMALIZATION LAYER
 62:   // =====================
 63:   private normalizeException(exception: unknown): {
 64:     status: HttpStatus;
 65:     code: string;
 66:     message: string;
 67:     details?: unknown;
 68:     stack?: string;
 69:   } {
 70:     let status: HttpStatus = HttpStatus.INTERNAL_SERVER_ERROR;
 71:     let code: string = ERROR_CODES.INTERNAL_SERVER_ERROR;
 72:     let message = 'An unexpected error occurred';
 73:     let details: unknown;
 74:     let stack: string | undefined;
 75: 
 76:     if (exception instanceof HttpException) {
 77:       status = exception.getStatus();
 78: 
 79:       const res: unknown = exception.getResponse();
 80: 
 81:       if (typeof res === 'string') {
 82:         message = res;
 83:       } else if (typeof res === 'object' && res !== null) {
 84:         const errObj = res as {
 85:           code?: string;
 86:           message?: string | string[];
 87:           details?: unknown;
 88:         };
 89: 
 90:         code = errObj.code ?? this.mapStatusToCode(status);
 91: 
 92:         message = Array.isArray(errObj.message)
 93:           ? errObj.message.join(', ')
 94:           : (errObj.message ?? exception.message);
 95: 
 96:         details = errObj.details;
 97:       }
 98:     } else if (exception instanceof Error) {
 99:       message = exception.message;
100:       stack = exception.stack;
101:     }
102: 
103:     // attach stack only in dev
104:     if (process.env.NODE_ENV !== 'production' && stack) {
105:       details = {
106:         ...(typeof details === 'object' && details ? details : {}),
107:         stack,
108:       };
109:     }
110: 
111:     return { status, code, message, details, stack };
112:   }
113: 
114:   // =====================
115:   // LOGGING
116:   // =====================
117:   private logException(
118:     request: Request,
119:     status: number,
120:     code: string,
121:     message: string,
122:     exception: unknown,
123:     stack?: string,
124:     details?: unknown,
125:   ) {
126:     const logPayload = {
127:       path: request.url,
128:       method: request.method,
129:       status,
130:       body: request.body as unknown,
131:       details,
132:     };
133: 
134:     const logMessage = `[${request.method}] ${request.url} - ${code}: ${message}`;
135: 
136:     if (status >= 500) {
137:       this.logger.error(
138:         logMessage,
139:         exception instanceof Error ? exception.stack : stack,
140:         JSON.stringify(logPayload),
141:       );
142:     } else {
143:       this.logger.warn(logMessage, undefined, JSON.stringify(logPayload));
144:     }
145:   }
146: 
147:   // =====================
148:   // STATUS MAP
149:   // =====================
150:   private mapStatusToCode(status: HttpStatus): string {
151:     switch (status) {
152:       case HttpStatus.BAD_REQUEST:
153:         return ERROR_CODES.VALIDATION_FAILED;
154:       case HttpStatus.UNAUTHORIZED:
155:         return ERROR_CODES.AUTH_INVALID_TOKEN;
156:       case HttpStatus.FORBIDDEN:
157:         return ERROR_CODES.AUTH_INSUFFICIENT_PERMISSIONS;
158:       case HttpStatus.NOT_FOUND:
159:         return ERROR_CODES.RESOURCE_NOT_FOUND;
160:       case HttpStatus.CONFLICT:
161:         return ERROR_CODES.DUPLICATE_ENTRY;
162:       case HttpStatus.TOO_MANY_REQUESTS:
163:         return ERROR_CODES.RATE_LIMIT_EXCEEDED;
164:       default:
165:         return ERROR_CODES.INTERNAL_SERVER_ERROR;
166:     }
167:   }
168: }

 3: /**
 4:  * Serialize an object safely, handling circular references and limiting string length.
 5:  * @param value - The value to serialize
 6:  * @param maxDepth - Maximum depth for nested objects (default 5)
 7:  * @returns Serialized safe object or primitive
 8:  */
 9: export function safeSerialize(value: unknown, maxDepth: number = 5): unknown {
10:   if (value === null || value === undefined) {
11:     return undefined;
12:   }
13: 
14:   if (typeof value !== 'object') {
15:     if (typeof value === 'string') {
16:       return value.slice(0, 1000); // Limit string length
17:     }
18:     return value;
19:   }
20: 
21:   const seen = new WeakSet<object>();
22:   const serialize = (obj: unknown, depth: number): unknown => {
23:     if (depth > maxDepth) {
24:       return { _truncated: true, _reason: 'max depth exceeded' };
25:     }
26: 
27:     if (obj === null || obj === undefined) {
28:       return undefined;
29:     }
30: 
31:     if (typeof obj !== 'object') {
32:       if (typeof obj === 'string') return obj.slice(0, 1000);
33:       return obj;
34:     }
35: 
36:     // Sau các kiểm tra, obj chắc chắn là object (không null/undefined)
37:     if (seen.has(obj)) {
38:       return '[Circular]';
39:     }
40:     seen.add(obj);
41: 
42:     if (Array.isArray(obj)) {
43:       return obj.map((item) => serialize(item, depth + 1));
44:     }
45: 
46:     const result: Record<string, unknown> = {};
47:     for (const [key, val] of Object.entries(obj)) {
