21:     if (this.state === 'OPEN') {
22:       const now = Date.now();
23:       if (this.lastFailureTime && now - this.lastFailureTime > this.timeout) {
24:         this.state = 'HALF_OPEN';
25:         this.logger.warn(
26:           `Circuit breaker ${this.name} is half-open, allowing one request`,
27:         );
28:       } else {

30:           `Circuit breaker ${this.name} is open, rejecting request`,
31:         );
32:         throw new Error(`Service ${this.name} is unavailable (circuit open)`);
33:       }
34:     }
35: 
36:     try {
37:       const result = await fn();
38:       this.onSuccess();
39:       return result;
40:     } catch (error) {
41:       this.onFailure();
42:       throw error;
43:     }
44:   }
45: 
46:   private onSuccess() {
47:     if (this.state === 'HALF_OPEN') {
48:       this.state = 'CLOSED';
49:       this.failureCount = 0;
50:       this.logger.log(
51:         `Circuit breaker ${this.name} closed after successful request`,
52:       );
53:     } else {
54:       this.failureCount = 0;
55:     }
56:   }
57: 
58:   private onFailure() {
59:     this.failureCount++;

65:       );

69:     ) {
70:       this.state = 'OPEN';

 61:       timeout: timeoutMs,
 62:     });
 63:

 65:       const ctx = requestContext.getStore();

 69:       return config;

 77:           _retryCount?: number;
 78:         };

 80:         config._retryCount = config._retryCount ?? 0;
 81:         if (config._retryCount < retries && this.shouldRetry(error)) {
 82:           config._retryCount++;
 83:           await this.delay(retryDelay * config._retryCount);
 84:           return this.client(config);
 85:         }

 87:       },
