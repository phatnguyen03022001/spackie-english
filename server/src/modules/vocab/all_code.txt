// src/modules/vocab/services/review.service.ts
import {
  Injectable,
  NotFoundException,
  BadRequestException,
} from '@nestjs/common';
import { PrismaService } from '@/modules/prisma/prisma.service';
import { UsersService } from '@/modules/users/users.service';
import {
  CardStatus,
  Prisma,
  LearningSession,
  Card,
  Deck,
} from '@prisma/client';
import { InjectPinoLogger, PinoLogger } from 'nestjs-pino';
import { SyncReviewDto } from '../dto/vocab.dto';

@Injectable()
export class ReviewService {
  constructor(
    private readonly prisma: PrismaService,
    private readonly usersService: UsersService,
    @InjectPinoLogger(ReviewService.name)
    private readonly logger: PinoLogger,
  ) {}

  async enrollDeck(userId: string, deckId: string) {
    const masterDeck = await this.prisma.deck.findUnique({
      where: { id: deckId },
      include: { cards: true },
    });

    if (!masterDeck || !masterDeck.isPublic) {
      throw new NotFoundException('Bộ thẻ không tồn tại hoặc chưa công khai');
    }

    const userExistingCards = await this.prisma.card.findMany({
      where: { userId },
      select: { word: true },
    });

    const existingWords = new Set(userExistingCards.map((c) => c.word));
    const newCards = masterDeck.cards.filter(
      (card) => !existingWords.has(card.word),
    );

    if (newCards.length === 0) {
      return { message: 'Bạn đã sở hữu tất cả từ', totalCards: 0 };
    }

    try {
      await this.prisma.card.createMany({
        data: newCards.map((card) => ({
          word: card.word,
          phonetic: card.phonetic,
          audioUrl: card.audioUrl,
          // Ép kiểu để Prisma hiểu đây là mảng Embedded Meaning
          meanings: card.meanings as unknown as Prisma.MeaningCreateInput[],
          userId,
          deckId,
          status: CardStatus.NEW,
          interval: 0,
          repetition: 0,
          easeFactor: 2.5,
          nextReview: new Date(),
        })),
      });

      return {
        message: `Đã thêm ${newCards.length} từ mới`,
        totalCards: newCards.length,
      };
    } catch (error) {
      this.logger.error(error as Error, 'Enroll deck failed');
      throw new BadRequestException('Không thể kích hoạt bộ thẻ.');
    }
  }

  async bulkSyncProgress(userId: string, dto: SyncReviewDto) {
    const { results, sessionId } = dto;
    if (!results.length) return { success: true, xpEarned: 0 };

    try {
      return await this.prisma.$transaction(
        async (tx: Prisma.TransactionClient) => {
          const logs: Prisma.ReviewLogCreateManyInput[] = [];

          for (const res of results) {
            // Lấy snapshot để lưu ReviewLog (Dùng cho tính năng Undo)
            const currentCard = (await tx.card.findUnique({
              where: { id: res.cardId, userId },
              select: {
                status: true,
                interval: true,
                easeFactor: true,
                nextReview: true,
              },
            })) as Pick<
              Card,
              'status' | 'interval' | 'easeFactor' | 'nextReview'
            > | null;

            if (currentCard) {
              logs.push({
                userId,
                cardId: res.cardId,
                sessionId: sessionId || null,
                rating: res.lastGrade || 3,
                responseTime: res.responseTime || 0,
                oldStatus: currentCard.status,
                oldInterval: currentCard.interval,
                oldEase: currentCard.easeFactor,
                oldNextReview: currentCard.nextReview,
                // Dữ liệu từ DTO
                newStatus: res.status,
                newInterval: res.interval,
                newEase: res.easeFactor,
                newNextReview: new Date(res.nextReview),
              });
            }

            // Cập nhật trạng thái Card theo SM-2 từ Client
            await Promise.all(
              results.map((res) =>
                tx.card.update({
                  where: { id: res.cardId },
                  data: {
                    status: res.status,
                    interval: res.interval,
                    repetition: res.repetition,
                    easeFactor: res.easeFactor,
                    nextReview: new Date(res.nextReview),
                    lastGrade: res.lastGrade,
                    lastReviewedAt: new Date(),
                  },
                }),
              ),
            );
          }

          // Ghi log hàng loạt
          if (logs.length > 0) {
            await tx.reviewLog.createMany({ data: logs });
          }

          // Cập nhật session nếu client gửi sessionId
          if (sessionId) {
            await tx.learningSession.update({
              where: { id: sessionId },
              data: {
                cardsLearned: { increment: results.length },
                endTime: new Date(),
              },
            });
          }

          const xpEarned = results.length * 10;
          await this.usersService.addXp(userId, xpEarned);

          return { success: true, xpEarned, cardsProcessed: results.length };
        },
      );
    } catch (error) {
      this.logger.error(error as Error, 'Bulk sync failed');
      throw new BadRequestException('Đồng bộ tiến độ thất bại.');
    }
  }

  async undoLastAction(userId: string) {
    // Xóa bỏ đoạn "as ReviewLog | null"
    const lastLog = await this.prisma.reviewLog.findFirst({
      where: { userId },
      orderBy: { createdAt: 'desc' },
    });

    // TypeScript tự hiểu lastLog có thể là null,
    // nên check này là bắt buộc và hợp lệ
    if (!lastLog) {
      throw new NotFoundException('Không có hành động nào để hoàn tác');
    }

    try {
      return await this.prisma.$transaction(
        async (tx: Prisma.TransactionClient) => {
          await tx.card.update({
            where: { id: lastLog.cardId },
            data: {
              status: lastLog.oldStatus,
              interval: lastLog.oldInterval,
              easeFactor: lastLog.oldEase,
              nextReview: lastLog.oldNextReview,
              lastReviewedAt: null,
            },
          });

          await tx.reviewLog.delete({ where: { id: lastLog.id } });
          return { success: true, message: 'Hoàn tác thành công' };
        },
      );
    } catch (error) {
      this.logger.error(error as Error, 'Undo failed');
      throw new BadRequestException('Không thể hoàn tác.');
    }
  }

  async createLearningSession(
    userId: string,
    deckId: string,
  ): Promise<LearningSession> {
    // Prisma đã biết chắc chắn kết quả trả về là LearningSession
    return await this.prisma.learningSession.create({
      data: {
        userId,
        deckId,
        startTime: new Date(),
      },
    });
  }

  async getTodayReviews(userId: string): Promise<Card[]> {
    const cards = await this.prisma.card.findMany({
      where: {
        userId,
        nextReview: { lte: new Date() },
        status: { not: CardStatus.NEW },
      },
      orderBy: [{ status: 'desc' }, { nextReview: 'asc' }],
    });

    return cards;
  }
  async getReviewForecast(userId: string): Promise<Record<string, number>> {
    const cards = await this.prisma.card.findMany({
      where: { userId, status: { not: CardStatus.NEW } },
      select: { nextReview: true },
    });

    const forecast: Record<string, number> = {};
    cards.forEach((card) => {
      const dateKey = card.nextReview.toISOString().split('T')[0];
      forecast[dateKey] = (forecast[dateKey] || 0) + 1;
    });
    return forecast;
  }

  async getEnrolledDecks(userId: string): Promise<Deck[]> {
    const cardDecks = await this.prisma.card.findMany({
      where: { userId, deckId: { not: null } },
      distinct: ['deckId'],
      select: { deckId: true },
    });

    const deckIds = cardDecks
      .map((c) => c.deckId)
      .filter((id): id is string => id !== null);

    return this.prisma.deck.findMany({
      where: { id: { in: deckIds } },
      include: { _count: { select: { cards: true } } },
    });
  }

  async unenrollDeck(userId: string, deckId: string) {
    await this.prisma.card.deleteMany({ where: { userId, deckId } });
    return { success: true };
  }

  async getRecommendedDecks(userId: string) {
    const enrolledDecks = await this.getEnrolledDecks(userId);
    const enrolledIds = enrolledDecks.map((d) => d.id);
    const tags = enrolledDecks
      .map((d) => d.levelTag)
      .filter((t): t is string => t !== null);

    return await this.prisma.deck.findMany({
      where: {
        isPublic: true,
        id: { notIn: enrolledIds },
        OR: [
          tags.length > 0 ? { levelTag: { in: tags } } : {},
          { isPublic: true },
        ],
      },
      take: 5,
      include: { _count: { select: { cards: true } } },
    });
  }
}

import { PrismaService } from '@/modules/prisma/prisma.service';

import { InjectPinoLogger, PinoLogger } from 'nestjs-pino';
import { ConfigService } from '@nestjs/config';
import { translate } from '@vitalets/google-translate-api';
import {
  MeaningDto,
  CreateDeckDto,
  CreateCardDto,
  UpdateDeckDto,
  UpdateCardDto,
} from '../dto/vocab.dto';

interface IDictionaryResponse {
  word: string;
  phonetic?: string;
  phonetics: { text?: string; audio?: string }[];
  meanings: {
    partOfSpeech: string;
    definitions: {
      definition: string;
      example?: string;
      synonyms?: string[];
      antonyms?: string[];
    }[];
  }[];
}

@Injectable()
export class ManagementService {
  constructor(
    private readonly prisma: PrismaService,
    private readonly configService: ConfigService,
    @InjectPinoLogger(ManagementService.name)
    private readonly logger: PinoLogger,
  ) {}

  async createDeck(creatorId: string, dto: CreateDeckDto) {
    return this.prisma.deck.create({
      data: { ...dto, creatorId },
    });
  }

  async updateDeckMetadata(id: string, dto: UpdateDeckDto) {
    const deck = await this.prisma.deck.findUnique({ where: { id } });
    if (!deck) throw new NotFoundException('Không tìm thấy bộ thẻ');

    return this.prisma.deck.update({
      where: { id },
      data: {
        ...dto,
        updatedAt: new Date(),
      },
    });
  }

  async updateDeckStatus(id: string, isPublic: boolean) {
    return this.prisma.deck.update({
      where: { id },
      data: { isPublic, updatedAt: new Date() },
    });
  }

  async softDeleteDeck(id: string) {
    // Thay vì xóa hoàn toàn Card, bạn có thể chỉ cần ngắt kết nối deckId
    // để User vẫn giữ được từ vựng cá nhân nếu muốn
    return this.prisma.$transaction([
      this.prisma.card.updateMany({
        where: { deckId: id },
        data: { deckId: null },
      }),
      this.prisma.deck.delete({ where: { id } }),
    ]);
  }

  async findTeacherDecks(teacherId: string) {
    return this.prisma.deck.findMany({
      where: { creatorId: teacherId },
      include: { _count: { select: { cards: true } } },
      orderBy: { createdAt: 'desc' },
    });
  }

  async findPublicDecks(search?: string, tag?: string) {
    return this.prisma.deck.findMany({
      where: {
        isPublic: true,
        AND: [
          search ? { title: { contains: search, mode: 'insensitive' } } : {},
          tag ? { levelTag: tag } : {},
        ],
      },
      include: {
        _count: { select: { cards: true } },
        creator: { select: { name: true } },
      },
      orderBy: { createdAt: 'desc' },
    });
  }

  async getDeckWithCards(id: string) {
    const deck = await this.prisma.deck.findUnique({
      where: { id },
      include: {
        cards: true,
        creator: { select: { name: true } },
      },
    });
    if (!deck) throw new NotFoundException('Không tìm thấy bộ thẻ');
    return deck;
  }

  async createCardManually(deckId: string, dto: CreateCardDto) {
    const deck = await this.prisma.deck.findUnique({
      where: { id: deckId },
      select: { creatorId: true },
    });
    if (!deck) throw new NotFoundException('Bộ thẻ không tồn tại');

    // Kiểm tra từ này đã có trong bộ thẻ chưa
    const existing = await this.prisma.card.findFirst({
      where: { deckId, word: dto.word.toLowerCase() },
    });
    if (existing)
      throw new BadRequestException('Từ vựng này đã tồn tại trong bộ thẻ');

    return this.prisma.card.create({
      data: {
        word: dto.word.toLowerCase(),
        phonetic: dto.phonetic,
        audioUrl: dto.audioUrl,
        meanings: dto.meanings as Prisma.MeaningCreateInput[],
        user: { connect: { id: deck.creatorId } },
        deck: { connect: { id: deckId } },
      },
    });
  }

  async bulkCreateCardsWithAutoFill(deckId: string, words: string[]) {
    if (!words?.length) throw new BadRequestException('Danh sách từ trống');

    const deck = await this.prisma.deck.findUnique({
      where: { id: deckId },
      select: { creatorId: true },
    });
    if (!deck) throw new NotFoundException('Không tìm thấy bộ thẻ');

    const uniqueWords = Array.from(
      new Set(words.map((w) => w.trim().toLowerCase())),
    ).filter(Boolean);

    const results = await Promise.all(
      uniqueWords.map(async (word) => {
        try {
          // Kiểm tra xem từ này đã tồn tại ở bất kỳ đâu trong hệ thống chưa (để tái sử dụng data)
          const existingGlobalCards = await this.prisma.card.findMany({
            where: { word: { in: uniqueWords } },
            distinct: ['word'],
          });
          const existingMap = new Map(
            existingGlobalCards.map((c) => [c.word, c]),
          );

          // Trong vòng lặp map:
          const globalData = existingMap.get(word);
          if (globalData) {
            return {
              word,
              phonetic: globalData.phonetic,
              audioUrl: globalData.audioUrl,
              meanings: globalData.meanings as unknown as MeaningDto[],
            };
          }

          return await this.getMergedData(word);
        } catch {
          this.logger.error({ word }, 'Auto-fill failed for word');
          return null;
        }
      }),
    );

    const validData = results.filter(
      (r): r is NonNullable<typeof r> => r !== null,
    );

    await this.prisma.$transaction(
      validData.map((data) =>
        this.prisma.card.create({
          data: {
            word: data.word,
            phonetic: data.phonetic,
            audioUrl: data.audioUrl,
            meanings: data.meanings as Prisma.MeaningCreateInput[],
            user: { connect: { id: deck.creatorId } },
            deck: { connect: { id: deckId } },
          },
        }),
      ),
    );

    return { success: true, addedCount: validData.length };
  }

  async updateCard(cardId: string, dto: UpdateCardDto) {
    const card = await this.prisma.card.findUnique({ where: { id: cardId } });
    if (!card) throw new NotFoundException('Không tìm thấy thẻ');

    return this.prisma.card.update({
      where: { id: cardId },
      data: {
        word: dto.word?.toLowerCase(),
        phonetic: dto.phonetic,
        audioUrl: dto.audioUrl,
        // FIX: Ép kiểu sang Prisma.MeaningCreateInput[] thay vì InputJsonValue
        meanings: dto.meanings
          ? (dto.meanings as unknown as Prisma.MeaningCreateInput[])
          : undefined,
        deckId: dto.deckId,
        status: dto.status,
        updatedAt: new Date(),
      },
    });
  }

  async deleteCard(cardId: string, userId: string) {
    // Nhận thêm userId từ Controller
    const card = await this.prisma.card.findFirst({
      where: { id: cardId, userId }, // Đảm bảo đúng chủ sở hữu
    });
    if (!card)
      throw new NotFoundException(
        'Card not found or you do not have authorization',
      );
    return this.prisma.card.delete({ where: { id: cardId } });
  }

  private async getMergedData(word: string) {
    const publicData = await this.fetchPublicDictionary(word);
    if (!publicData) return null;

    const viTranslation = await this.translateToVietnamese(word);
    const meanings: MeaningDto[] = [];

    if (viTranslation) {
      meanings.push({
        partOfSpeech: 'Vietnamese',
        definitions: [{ definition: viTranslation }],
      });
    }

    if (publicData.meanings?.length > 0) {
      const firstMeaning = publicData.meanings[0];
      meanings.push({
        partOfSpeech: firstMeaning.partOfSpeech,
        definitions: firstMeaning.definitions.slice(0, 2).map((d) => ({
          definition: d.definition,
          example: d.example,
          synonyms: d.synonyms || [],
          antonyms: d.antonyms || [],
        })),
      });
    }

    return {
      word: publicData.word.toLowerCase(),
      phonetic:
        publicData.phonetic ||
        publicData.phonetics.find((p) => p.text)?.text ||
        null,
      audioUrl: publicData.phonetics.find((p) => p.audio)?.audio || null,
      meanings,
    };
  }

  private async translateToVietnamese(word: string): Promise<string | null> {
    try {
      const res = await translate(word, { to: 'vi' });
      return res.text;
    } catch {
      return null;
    }
  }

  private async fetchPublicDictionary(
    word: string,
  ): Promise<IDictionaryResponse | null> {
    try {
      const data = await typedAxiosGet<IDictionaryResponse[]>(
        `https://api.dictionaryapi.dev/api/v2/entries/en/${encodeURIComponent(word)}`,
      );
      return data?.[0] || null;
    } catch {
      return null;
    }
  }
}

import { Injectable } from '@nestjs/common';
import { PrismaService } from '@/modules/prisma/prisma.service';
import { CardStatus } from '@prisma/client';

@Injectable()
export class AnalyticsService {
  constructor(private readonly prisma: PrismaService) {}

  /**
   * 1. Thống kê trạng thái thẻ cho Dashboard
   */
  async getUserDashboardStats(userId: string) {
    const [statusGroups, totalCards] = await Promise.all([
      this.prisma.card.groupBy({
        by: ['status'],
        where: { userId },
        _count: { status: true },
      }),
      this.prisma.card.count({ where: { userId } }),
    ]);

    const statusStats: Record<CardStatus, number> = {
      [CardStatus.NEW]: 0,
      [CardStatus.LEARNING]: 0,
      [CardStatus.REVIEW]: 0,
      [CardStatus.LAPSED]: 0,
    };

    statusGroups.forEach((group) => {
      const status = group.status;
      if (status) {
        statusStats[status] = group._count.status;
      }
    });

    return {
      totalCards,
      statusStats,
      masteryRate:
        totalCards > 0
          ? Math.round((statusStats[CardStatus.REVIEW] / totalCards) * 100)
          : 0,
    };
  }

  /**
   * 2. Lấy danh sách các bộ thẻ học hiệu quả nhất (nhiều thẻ đã thuộc nhất)
   */
  async getTopPerformingDecks(userId: string, limit = 3) {
    const deckStats = await this.prisma.card.groupBy({
      by: ['deckId'],
      where: {
        userId,
        status: CardStatus.REVIEW,
        deckId: { not: null }, // Đảm bảo không lấy thẻ lẻ không thuộc deck
      },
      _count: {
        deckId: true,
      },
      orderBy: {
        _count: {
          deckId: 'desc',
        },
      },
      take: limit,
    });

    const deckIds = deckStats
      .map((s) => s.deckId)
      .filter((id): id is string => id !== null);

    const decks = await this.prisma.deck.findMany({
      where: { id: { in: deckIds } },
      select: { id: true, title: true },
    });

    return (
      deckStats
        .map((stat) => {
          const deckInfo = decks.find((d) => d.id === stat.deckId);
          return {
            // XÓA "as string" ở đây
            id: stat.deckId,
            title: deckInfo?.title ?? 'Unknown Deck',
            masteredCount: stat._count.deckId,
          };
        })
        // Type Guard này giúp TS chắc chắn 100% id không phải null
        .filter(
          (d): d is { id: string; title: string; masteredCount: number } =>
            d.id !== null,
        )
    );
  }

  /**
   * 3. Lấy dữ liệu biểu đồ nhiệt (Heatmap) - FIX LỖI TS2339
   */
  async getHeatmap(userId: string) {
    const thirtyDaysAgo = new Date();
    thirtyDaysAgo.setDate(thirtyDaysAgo.getDate() - 30);

    // Truy vấn số lượng bản ghi log theo ngày
    const logs = await this.prisma.reviewLog.findMany({
      where: {
        userId,
        createdAt: { gte: thirtyDaysAgo },
      },
      select: { createdAt: true },
    });

    const heatmap: Record<string, number> = {};

    logs.forEach((log) => {
      // Định dạng date theo YYYY-MM-DD
      const dateKey = log.createdAt.toISOString().split('T')[0];
      heatmap[dateKey] = (heatmap[dateKey] || 0) + 1;
    });

    return heatmap;
  }
